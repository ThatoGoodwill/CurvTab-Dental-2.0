using CurvTabDental.API.Notifications;
using System.Threading.RateLimiting;
using CurvTabDental.API.Middleware;
using CurvTabDental.API.Security;
using CurvTabDental.API.Seeding;
using CurvTabDental.Application.Abstractions;
using CurvTabDental.Infrastructure.Identity;
using CurvTabDental.Infrastructure.Persistence;
using CurvTabDental.Infrastructure.Services;
using Microsoft.AspNetCore.Authorization;
using Microsoft.AspNetCore.Identity;
using Microsoft.EntityFrameworkCore;

var builder = WebApplication.CreateBuilder(args);

// ---------- Core services ----------
builder.Services.AddControllers();
builder.Services.AddSignalR();
builder.Services.AddOpenApi();
builder.Services.AddProblemDetails();
builder.Services.AddHttpContextAccessor();

builder.Services.AddScoped<ICurrentUser, HttpCurrentUser>();
builder.Services.AddScoped<IAuditService, AuditService>();

// ---------- Database ----------
// ConnectionStrings:DentalDatabase lives in dotnet user-secrets (never in Git).
builder.Services.AddDbContext<DentalDbContext>(options =>
    options.UseNpgsql(builder.Configuration.GetConnectionString("DentalDatabase")));

// ---------- Identity + cookie authentication ----------
builder.Services
    .AddIdentity<ApplicationUser, ApplicationRole>(options =>
    {
        options.User.RequireUniqueEmail = true;

        options.Password.RequiredLength = 10;
        options.Password.RequireDigit = true;
        options.Password.RequireLowercase = true;
        options.Password.RequireUppercase = true;
        options.Password.RequireNonAlphanumeric = false;

        options.Lockout.MaxFailedAccessAttempts = 5;
        options.Lockout.DefaultLockoutTimeSpan = TimeSpan.FromMinutes(15);
        options.Lockout.AllowedForNewUsers = true;
    })
    .AddEntityFrameworkStores<DentalDbContext>()
    .AddClaimsPrincipalFactory<PracticeClaimsPrincipalFactory>()
    .AddDefaultTokenProviders();

builder.Services.ConfigureApplicationCookie(options =>
{
    options.Cookie.Name = "ctd.auth";
    options.Cookie.HttpOnly = true;
    options.Cookie.SameSite = SameSiteMode.Lax;
    options.Cookie.SecurePolicy = builder.Environment.IsDevelopment()
        ? CookieSecurePolicy.SameAsRequest
        : CookieSecurePolicy.Always;
    options.ExpireTimeSpan = TimeSpan.FromHours(8);
    options.SlidingExpiration = true;

    // This is an API: answer 401/403 instead of redirecting to a login page.
    options.Events.OnRedirectToLogin = context =>
    {
        context.Response.StatusCode = StatusCodes.Status401Unauthorized;
        return Task.CompletedTask;
    };
    options.Events.OnRedirectToAccessDenied = context =>
    {
        context.Response.StatusCode = StatusCodes.Status403Forbidden;
        return Task.CompletedTask;
    };
});

// Deny by default (FR-009): any endpoint without [AllowAnonymous] requires a signed-in user.
builder.Services.AddAuthorization(options =>
{
    options.FallbackPolicy = new AuthorizationPolicyBuilder()
        .RequireAuthenticatedUser()
        .Build();
});

// ---------- CORS for the React front-end ----------
var allowedOrigins = builder.Configuration.GetSection("Cors:AllowedOrigins").Get<string[]>()
                     ?? new[] { "http://localhost:5173" };
builder.Services.AddCors(options =>
{
    options.AddPolicy("Frontend", policy => policy
        .WithOrigins(allowedOrigins)
        .AllowAnyHeader()
        .AllowAnyMethod()
        .AllowCredentials());
});

// ---------- Rate limiting (login brute-force; lockout is the second layer) ----------
builder.Services.AddRateLimiter(options =>
{
    options.RejectionStatusCode = StatusCodes.Status429TooManyRequests;
    options.AddPolicy("login", httpContext =>
        RateLimitPartition.GetFixedWindowLimiter(
            httpContext.Connection.RemoteIpAddress?.ToString() ?? "unknown",
            _ => new FixedWindowRateLimiterOptions
            {
                PermitLimit = 20,
                Window = TimeSpan.FromMinutes(1),
                QueueLimit = 0
            }));
});

var app = builder.Build();

// ---------- Seed roles / bootstrap admin / demo data ----------
try
{
    await IdentitySeeder.SeedAsync(app.Services, app.Configuration, app.Environment);
}
catch (Exception ex)
{
    app.Logger.LogError(ex,
        "Seeding failed. Have you applied the migrations? (dotnet ef database update ...)");
}

// ---------- HTTP pipeline (order matters) ----------
app.UseExceptionHandler();

if (app.Environment.IsDevelopment())
{
    app.MapOpenApi();
}
else
{
    app.UseHsts();
    app.UseHttpsRedirection();
}

app.UseCors("Frontend");
app.UseRateLimiter();
app.UseMiddleware<CsrfHeaderMiddleware>();
app.UseAuthentication();
app.UseMiddleware<ActiveUserMiddleware>();
app.UseAuthorization();

app.MapHub<ClinicNotificationsHub>("/api/hubs/clinic");
app.MapControllers();

app.Run();

/// <summary>
/// One database for Identity + all practice data.
/// Tenant isolation: every IPracticeOwned entity gets a global query filter comparing its
/// DentalPracticeId with the practice of the signed-in user (from ICurrentUser).
/// A platform user (no practice) therefore sees ZERO practice-owned rows.
/// </summary>
public class DentalDbContext : IdentityDbContext<ApplicationUser, ApplicationRole, Guid>

    }

    [HttpGet]

        [FromQuery] string? q,

        [FromQuery] int page = 1,
        [FromQuery] int pageSize = 25,
        CancellationToken ct = default)
    {
        page = Math.Max(page, 1);
        pageSize = Math.Clamp(pageSize, 1, 100);

        IQueryable<Patient> query = _db.Patients.AsNoTracking();
        if (!includeArchived) query = query.Where(p => !p.IsArchived);

        if (!string.IsNullOrWhiteSpace(q))
        {
            var pattern = "%" + EscapeLike(q.Trim()) + "%";
            query = query.Where(p =>
                EF.Functions.ILike(p.FirstName, pattern) ||
                EF.Functions.ILike(p.LastName, pattern) ||
                (p.PhoneNumber != null && EF.Functions.ILike(p.PhoneNumber, pattern)) ||
                (p.IdNumber != null && EF.Functions.ILike(p.IdNumber, pattern)));
        }

        var total = await query.CountAsync(ct);
        var rows = await query
            .OrderBy(p => p.LastName).ThenBy(p => p.FirstName)
            .Skip((page - 1) * pageSize)
            .Take(pageSize)
            .ToListAsync(ct);

        var items = rows.Select(p => ToDto(p, _current.Role)).ToList();
        return Ok(new PagedResult<PatientDto>(items, total, page, pageSize));
    }

        return Ok(ToDto(patient, _current.Role));
    }

    [HttpPost]
    [Authorize(Roles = StaffRoles.PatientWrite)]
    public async Task<IActionResult> Create(
        [FromBody] PatientRequest request,
        [FromQuery] bool allowDuplicate = false,
        CancellationToken ct = default)
    {
        if (_current.PracticeId is not Guid practiceId) return Forbid();

        if (!allowDuplicate)
        {
            var duplicates = await FindDuplicatesAsync(request, excludeId: null, ct);
            if (duplicates.Count > 0) return DuplicateConflict(duplicates);
        }

        var now = DateTime.UtcNow;
        var patient = new Patient
        {
            DentalPracticeId = practiceId, // from the signed-in user, never from the request
            CreatedByUserId = _current.UserId,
            CreatedAtUtc = now,
            UpdatedAtUtc = now
        };
        Apply(patient, request, now);

        _db.Patients.Add(patient);
        await _db.SaveChangesAsync(ct);

        await _audit.WriteAsync("PATIENT_CREATED", "Patient", patient.Id.ToString(), ct: ct);
        return CreatedAtAction(nameof(Get), new { id = patient.Id }, ToDto(patient, _current.Role));
    }

    [HttpPut("{id:guid}")]
    [Authorize(Roles = StaffRoles.PatientWrite)]
    public async Task<IActionResult> Update(
        Guid id,
        [FromBody] PatientRequest request,
        [FromQuery] bool allowDuplicate = false,
        CancellationToken ct = default)
    {
        var patient = await _db.Patients.FirstOrDefaultAsync(p => p.Id == id, ct);
        if (patient is null) return NotFound();

        if (!allowDuplicate)
        {
            var duplicates = await FindDuplicatesAsync(request, excludeId: id, ct);
            if (duplicates.Count > 0) return DuplicateConflict(duplicates);
        }

        Apply(patient, request, DateTime.UtcNow);
        await _db.SaveChangesAsync(ct);

        await _audit.WriteAsync("PATIENT_UPDATED", "Patient", id.ToString(), ct: ct);
        return Ok(ToDto(patient, _current.Role));
    }

    [HttpPost("{id:guid}/archive")]
    [Authorize(Roles = StaffRoles.PatientWrite)]
    public async Task<IActionResult> Archive(Guid id, CancellationToken ct)
    {
        var patient = await _db.Patients.FirstOrDefaultAsync(p => p.Id == id, ct);
        if (patient is null) return NotFound();

        patient.IsArchived = true;
        patient.UpdatedAtUtc = DateTime.UtcNow;
        await _db.SaveChangesAsync(ct);

        await _audit.WriteAsync("PATIENT_ARCHIVED", "Patient", id.ToString(), ct: ct);
        return NoContent();
    }
